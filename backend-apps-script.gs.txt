 * ===== CẦN CÓ 3 SHEET SAU (đúng tên) =====
 * - "HSK": VOCAB_ID, VOCAB, PINYIN, PART-OF-SPEECH, MEANING, EXAMPLE,

var SPREADSHEET_ID = '183NqZLUSQvN81Vuni4sBLKy1f7g4VVFSWYToYuH5rhc';
var HSK_SHEET_NAME = 'HSK';
var GIAOTIEP_SHEET_NAME = 'GIAO TIẾP';
var PROGRESS_SHEET_NAME = 'PROGRESS';
var CONFIG_SHEET_NAME = 'CONFIG';
var MASTERY_MAX = 5;

function _ss() {
  return SpreadsheetApp.openById(SPREADSHEET_ID);
}

function _json(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

function _checkPassword(pw) {
  try {
    var cfg = _ss().getSheetByName(CONFIG_SHEET_NAME);
    if (!cfg) return true;
    var expected = cfg.getRange('A1').getValue();
    expected = (expected === null || expected === undefined) ? '' : String(expected).trim();
    if (!expected) return true;
    return String(pw || '').trim() === expected;
  } catch (err) {
    return true;
  }
}

// Đọc 1 sheet dạng bảng (dòng 1 = header) thành mảng object {header: value}
function _sheetToRows(sheet) {
  if (!sheet) return { headers: [], rows: [] };
  var values = sheet.getDataRange().getValues();
  if (values.length === 0) return { headers: [], rows: [] };
  var headers = values[0];
  var rows = [];
  for (var i = 1; i < values.length; i++) {
    var row = values[i];
    if (row.join('') === '') continue;
    var obj = {};
    headers.forEach(function (h, idx) { obj[h] = row[idx]; });
    obj._row = i + 1;
    rows.push(obj);
  }
  return { headers: headers, rows: rows };
}

// Đọc sheet PROGRESS thành map ID -> OK (số)
function _getProgressMap() {
  var sheet = _ss().getSheetByName(PROGRESS_SHEET_NAME);
  var map = {};
  if (!sheet) return map;
  var values = sheet.getDataRange().getValues();
  for (var i = 1; i < values.length; i++) {
    var id = values[i][0];
    if (id === '' || id === null || id === undefined) continue;
    var ok = values[i][1];
    var okNum = (ok === '' || ok === null || isNaN(ok)) ? 0 : Number(ok);
    map[String(id).trim()] = okNum;
  }
  return map;
}

function doGet(e) {
  try {
    var ss = _ss();
    var hsk = _sheetToRows(ss.getSheetByName(HSK_SHEET_NAME));
    var giaoTiep = _sheetToRows(ss.getSheetByName(GIAOTIEP_SHEET_NAME));
    var progress = _getProgressMap();
    return _json({
      ok: true,
      hskHeaders: hsk.headers,
      hskRows: hsk.rows,
      giaoTiepHeaders: giaoTiep.headers,
      giaoTiepRows: giaoTiep.rows,
      progress: progress,
      masteryMax: MASTERY_MAX,
    });
  } catch (err) {
    return _json({ ok: false, error: String(err) });
  }
}

// Tăng điểm OK trong sheet PROGRESS theo ID (VOCAB_ID hoặc SENT_ID).
// Chưa có dòng nào ứng với ID đó thì tự thêm dòng mới (OK=1).
function doPost(e) {
  try {
    var body = JSON.parse(e.postData.contents);
    if (!_checkPassword(body.password)) {
      return _json({ ok: false, authError: true, error: 'Sai mật khẩu' });
    }
    var action = body.action;

    if (action === 'incrementMastery') {
      var id = String(body.id || '').trim();
      if (!id) throw new Error('Thiếu ID');
      var sheet = _ss().getSheetByName(PROGRESS_SHEET_NAME);
      if (!sheet) throw new Error('Không tìm thấy sheet PROGRESS');

      var values = sheet.getDataRange().getValues();
      var foundRow = -1;
      for (var i = 1; i < values.length; i++) {
        if (String(values[i][0]).trim() === id) { foundRow = i + 1; break; }
      }

      var newVal;
      if (foundRow === -1) {
        newVal = 1;
        sheet.appendRow([id, newVal]);
      } else {
        var cur = values[foundRow - 1][1];
        var curNum = (cur === '' || cur === null || isNaN(cur)) ? 0 : Number(cur);
        newVal = curNum + 1;
        sheet.getRange(foundRow, 2).setValue(newVal);
      }
      return _json({ ok: true, id: id, newValue: newVal });
    }

    return _json({ ok: false, error: 'Hành động không xác định: ' + action });
  } catch (err) {
    return _json({ ok: false, error: String(err) });
  }
}

function onOpen() {
  SpreadsheetApp.getUi()
    .createMenu('🈶 Công cụ ALIGN')
    .addItem('Tự động tách ALIGN (Giao Tiếp)', 'autoAlignGiaoTiep')
    .addItem('Tự động tách ALIGN (HSK Example)', 'autoAlignHskExample')
    .addSeparator()
    .addItem('Tạo khung ALIGN từ cột đếm (Giao Tiếp)', 'generateAlignTemplateGiaoTiep')
    .addItem('Tạo khung ALIGN từ cột đếm (HSK Example)', 'generateAlignTemplateHskExample')
    .addToUi();
}

// Xây từ điển {chữ Hán: nghĩa tiếng Việt} từ sheet HSK, sắp xếp theo độ dài
// giảm dần để khớp từ dài nhất trước (vd ưu tiên "认识" trước "认").
function _buildHskDict() {
  var sheet = _ss().getSheetByName(HSK_SHEET_NAME);
  var values = sheet.getDataRange().getValues();
  var headers = values[0];
  var vocabCol = headers.indexOf('VOCAB');
  var meaningCol = headers.indexOf('MEANING');
  if (vocabCol === -1 || meaningCol === -1) throw new Error('Sheet HSK thiếu cột VOCAB hoặc MEANING');

  var dict = {};
  for (var i = 1; i < values.length; i++) {
    var word = values[i][vocabCol];
    var meaning = values[i][meaningCol];
    if (!word || !meaning) continue;
    word = String(word).split('|')[0].trim(); // bỏ các biến thể "a | b", chỉ lấy dạng chính
    if (!word || dict[word]) continue;
    dict[word] = String(meaning).split(/[;,\/]/)[0].trim(); // lấy nghĩa đầu tiên nếu liệt kê nhiều nghĩa
  }
  var wordsSortedByLength = Object.keys(dict).sort(function (a, b) { return b.length - a.length; });
  return { dict: dict, words: wordsSortedByLength };
}

// Tách 1 câu thành các cụm dựa theo từ điển (khớp dài nhất trước).
// Ký tự không khớp được từ nào (dấu câu, chữ hiếm...) sẽ bị bỏ qua.
function _segmentSentence(sentence, dictData) {
  var dict = dictData.dict;
  var words = dictData.words;
  var chunks = [];
  var i = 0;
  while (i < sentence.length) {
    var matched = null;
    for (var w = 0; w < words.length; w++) {
      var word = words[w];
      if (word.length <= sentence.length - i && sentence.substr(i, word.length) === word) {
        matched = word;
        break;
      }
    }
    if (matched) {
      chunks.push({ zh: matched, vi: dict[matched] });
      i += matched.length;
    } else {
      i += 1; // bỏ qua 1 ký tự không nhận diện được
    }
  }
  return chunks;
}

// Xuất theo định dạng "đếm số ký tự" (vd: "1=được;2=tổng cộng;4=35 tệ")
// thay vì gõ lại cả cụm chữ Hán - gọn hơn và app đọc lại chắc chắn khớp.
function _chunksToAlignString(chunks) {
  return chunks.map(function (c) { return c.zh.length + '=' + c.vi; }).join(';');
}

function autoAlignGiaoTiep() {
  var sheet = _ss().getSheetByName(GIAOTIEP_SHEET_NAME);
  if (!sheet) throw new Error('Không tìm thấy sheet GIAO TIẾP');
  var dictData = _buildHskDict();

  var values = sheet.getDataRange().getValues();
  var headers = values[0];
  var chnCol = headers.indexOf('CHN');
  var alignCol = headers.indexOf('ALIGN');
  if (chnCol === -1) throw new Error('Sheet GIAO TIẾP thiếu cột CHN');
  if (alignCol === -1) {
    alignCol = headers.length;
    sheet.getRange(1, alignCol + 1).setValue('ALIGN');
  }

  var filled = 0;
  for (var r = 1; r < values.length; r++) {
    if (values[r][alignCol]) continue; // đã có sẵn, không đè
    var sentence = values[r][chnCol];
    if (!sentence) continue;
    var chunks = _segmentSentence(String(sentence), dictData);
    if (chunks.length === 0) continue;
    sheet.getRange(r + 1, alignCol + 1).setValue(_chunksToAlignString(chunks));
    filled++;
  }
  SpreadsheetApp.getUi().alert('Đã tự động điền ' + filled + ' dòng ALIGN (Giao Tiếp). Nhớ rà lại và sửa nghĩa cho khớp câu dịch thật.');
}

function autoAlignHskExample() {
  var sheet = _ss().getSheetByName(HSK_SHEET_NAME);
  if (!sheet) throw new Error('Không tìm thấy sheet HSK');
  var dictData = _buildHskDict();

  var values = sheet.getDataRange().getValues();
  var headers = values[0];
  var exampleCol = headers.indexOf('EXAMPLE');
  var alignCol = headers.indexOf('EXAMPLE ALIGN');
  if (exampleCol === -1) throw new Error('Sheet HSK thiếu cột EXAMPLE');
  if (alignCol === -1) {
    alignCol = headers.length;
    sheet.getRange(1, alignCol + 1).setValue('EXAMPLE ALIGN');
  }

  var filled = 0;
  for (var r = 1; r < values.length; r++) {
    if (values[r][alignCol]) continue;
    var sentence = values[r][exampleCol];
    if (!sentence) continue;
    var chunks = _segmentSentence(String(sentence), dictData);
    if (chunks.length === 0) continue;
    sheet.getRange(r + 1, alignCol + 1).setValue(_chunksToAlignString(chunks));
    filled++;
  }
  SpreadsheetApp.getUi().alert('Đã tự động điền ' + filled + ' dòng EXAMPLE ALIGN (HSK). Nhớ rà lại và sửa nghĩa cho khớp câu dịch thật.');
}

/**
 * ================================================================
 * TẠO KHUNG ALIGN TỪ 1 CỘT ĐẾM DO NGƯỜI DÙNG TỰ GÕ
 * ================================================================
 * Cách dùng:
 * 1. Tự thêm 1 cột tên đúng là "ALIGN_COUNTS" vào sheet GIAO TIẾP (và/hoặc
 *    sheet HSK nếu muốn dùng cho EXAMPLE). Đổi ALIGN_COUNTS_COL_NAME bên
 *    dưới nếu bạn muốn đặt tên cột khác.
 * 2. Ở mỗi dòng, gõ số ký tự từng cụm cách nhau bởi dấu "-", kết thúc bằng
 *    ";" — ví dụ: 1-2-4;  hoặc  2-3-1;
 * 3. Vào menu "🈶 Công cụ ALIGN" > "Tạo khung ALIGN từ cột đếm (...)"
 * 4. Cột ALIGN / EXAMPLE ALIGN sẽ được ĐIỀN SẴN (giá trị thường, không phải
 *    công thức) dạng: 1=; 2=; 4=;  — chỉ điền vào dòng đang TRỐNG, không đè
 *    lên dữ liệu đã có.
 * 5. Bạn tự gõ nghĩa tiếng Việt vào ngay sau từng dấu "=" là xong, ví dụ:
 *    1=được; 2=tổng cộng; 4=35 tệ;
 */
var ALIGN_COUNTS_COL_NAME = 'ALIGN_COUNTS'; // đổi tên này nếu cột đếm của bạn đặt tên khác

// "1-2-4;" hoặc "1-2-4" -> "1=; 2=; 4=;"
function _expandAlignTemplate(countsStr) {
  if (!countsStr) return '';
  var cleaned = countsStr.toString().trim().replace(/;+\s*$/, '');
  var parts = cleaned.split('-')
    .map(function (s) { return s.trim(); })
    .filter(function (s) { return s !== '' && /^\d+$/.test(s); });
  if (parts.length === 0) return '';
  return parts.map(function (n) { return n + '=;'; }).join(' ');
}

function _generateAlignTemplateForSheet(sheetName, alignColName) {
  var sheet = _ss().getSheetByName(sheetName);
  if (!sheet) throw new Error('Không tìm thấy sheet ' + sheetName);
  var values = sheet.getDataRange().getValues();
  var headers = values[0];
  var countsCol = headers.indexOf(ALIGN_COUNTS_COL_NAME);
  var alignCol = headers.indexOf(alignColName);
  if (countsCol === -1) throw new Error('Không tìm thấy cột ' + ALIGN_COUNTS_COL_NAME + ' trong sheet ' + sheetName + '. Hãy tự thêm cột này trước.');
  if (alignCol === -1) {
    alignCol = headers.length;
    sheet.getRange(1, alignCol + 1).setValue(alignColName);
  }

  var filled = 0;
  for (var r = 1; r < values.length; r++) {
    if (values[r][alignCol]) continue; // đã có sẵn, không đè
    var countsStr = values[r][countsCol];
    if (!countsStr) continue;
    var template = _expandAlignTemplate(countsStr);
    if (!template) continue;
    sheet.getRange(r + 1, alignCol + 1).setValue(template);
    filled++;
  }
  SpreadsheetApp.getUi().alert('Đã tạo khung ALIGN cho ' + filled + ' dòng. Giờ vào cột ' + alignColName + ' điền nghĩa tiếng Việt sau từng dấu "=".');
}

function generateAlignTemplateGiaoTiep() {
  _generateAlignTemplateForSheet(GIAOTIEP_SHEET_NAME, 'ALIGN');
}

function generateAlignTemplateHskExample() {
  _generateAlignTemplateForSheet(HSK_SHEET_NAME, 'EXAMPLE ALIGN');
}
